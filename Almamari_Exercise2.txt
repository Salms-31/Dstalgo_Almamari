        private void button1_Click(object sender, EventArgs e)
        {
            int number = Convert.ToInt32(txtPosNeg.Text);


            if (number > 0)
            {
                lblResult.Text = "Number is Positive.";
            }
            else if (number < 0)
            {
                lblResult.Text = "Number is Negative.";
            }
            else
            {
                lblResult.Text = "Number is 0.";
            }
        }
    }
