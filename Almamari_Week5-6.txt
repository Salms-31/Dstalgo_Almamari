        private void btnBS_Click(object sender, EventArgs e)
        {
            int[] binary = { 10, 20, 30, 40, 50, 60, 70 };
            int number = int.Parse(txtBS.Text);

            int left = 0;
            int right = binary.Length - 1;
            while (left <= right)
            {
                int mid = left + (right - left) / 2;
                if (binary[mid] == number)
                {
                    lblBS.Text = $"Status: {number} found at index {mid}.";
                    return;
                }
                else if (binary[mid] < number)
                {
                    left = mid + 1;
                }
                else
                {
                    right = mid - 1;
                }
            }
        }

        private void btnDivide_Click(object sender, EventArgs e)
        {
            int n = int.Parse(txtDivide.Text);
            int loop = 0;
            while (n > 1)
            {
                n /= 2;
                loop++;
            }
            lblDivide.Text = "Answer: " + n + ", Looped " + loop + " times";
        }

        private void btnBSSG_Click(object sender, EventArgs e)
        {
            int high = int.Parse(txtBSSG.Text);
            int low = 1;
            string result = "";

            while (low <= high)
            {
                int middle = (low + high) / 2;
                result += middle + "\n";
                high = middle - 1;
            }
            lblBSSG.Text = result;
        }
    }
}
